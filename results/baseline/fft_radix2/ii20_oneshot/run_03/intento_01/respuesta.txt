```cpp
#include <hls_math.h>

// Kernel: mariposa FFT radix-2 (Cooley-Tukey), N=1024 puntos complejos
// Firma obligatoria:
void fft_radix2(
    const float x_real[1024],
    const float x_imag[1024],
    float y_real[1024],
    float y_imag[1024])
{
    const int N = 1024;
    const int LOG2N = 10;

    // Buffers internos de trabajo (in-place durante las etapas de la FFT)
    float real[N];
    float imag[N];
#pragma HLS ARRAY_PARTITION variable=real complete dim=1
#pragma HLS ARRAY_PARTITION variable=imag complete dim=1

    // Tablas de twiddle factors (coseno/seno) precomputadas
    float twcos[N / 2];
    float twsin[N / 2];
#pragma HLS ARRAY_PARTITION variable=twcos complete dim=1
#pragma HLS ARRAY_PARTITION variable=twsin complete dim=1

    const float PI_F = 3.14159265358979323846f;

    // --- Generacion de twiddle factors ---
twiddle_gen:
    for (int k = 0; k < N / 2; k++) {
#pragma HLS PIPELINE II=1
        float angle = -2.0f * PI_F * (float)k / (float)N;
        twcos[k] = hls::cosf(angle);
        twsin[k] = hls::sinf(angle);
    }

    // --- Reordenamiento bit-reversal de la entrada ---
bitrev:
    for (int i = 0; i < N; i++) {
#pragma HLS PIPELINE II=1
        int rev = 0;
        int v = i;
        for (int b = 0; b < LOG2N; b++) {
#pragma HLS UNROLL
            rev = (rev << 1) | (v & 1);
            v >>= 1;
        }
        real[i] = x_real[rev];
        imag[i] = x_imag[rev];
    }

    // --- Etapas iterativas de Cooley-Tukey (decimation-in-time) ---
stage_loop:
    for (int s = 1; s <= LOG2N; s++) {
        int size = 1 << s;
        int halfsize = size >> 1;
        int tablestep = N / size;

    group_loop:
        for (int i = 0; i < N; i += size) {
#pragma HLS LOOP_TRIPCOUNT min=1 max=512

        butterfly_loop:
            for (int pos = 0; pos < halfsize; pos++) {
#pragma HLS PIPELINE II=1
#pragma HLS LOOP_TRIPCOUNT min=1 max=512
                int j = i + pos;
                int jh = j + halfsize;
                int k = pos * tablestep;

                float wr = twcos[k];
                float wi = twsin[k];

                float xr = real[jh];
                float xi = imag[jh];

                float tr = xr * wr - xi * wi;
                float ti = xr * wi + xi * wr;

                float ur = real[j];
                float ui = imag[j];

                real[j]  = ur + tr;
                imag[j]  = ui + ti;
                real[jh] = ur - tr;
                imag[jh] = ui - ti;
            }
        }
    }

    // --- Escritura de resultados ---
output_loop:
    for (int i = 0; i < N; i++) {
#pragma HLS PIPELINE II=1
        y_real[i] = real[i];
        y_imag[i] = imag[i];
    }
}
```
