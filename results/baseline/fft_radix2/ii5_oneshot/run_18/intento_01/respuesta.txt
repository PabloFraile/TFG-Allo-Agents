#include <cmath>

// ---------------------------------------------------------------------
// FFT radix-2 (Cooley-Tukey, decimation-in-time), N = 1024 puntos.
// Convención: X[k] = sum_n x[n] * exp(-j*2*pi*k*n/N)  (igual que numpy.fft.fft)
// ---------------------------------------------------------------------

static const int   FFT_N   = 1024;
static const int   FFT_LOGN = 10;          // log2(1024)
static const float PI_F    = 3.14159265358979323846f;

// Inversion de bits de un indice de FFT_LOGN bits (se infiere como logica
// combinacional, completamente desenrollada).
static inline int bit_reverse(int x, int bits) {
#pragma HLS INLINE
    int r = 0;
BITREV_UNROLL:
    for (int i = 0; i < bits; i++) {
#pragma HLS UNROLL
        r = (r << 1) | (x & 1);
        x >>= 1;
    }
    return r;
}

void fft_radix2(
    const float x_real[1024],
    const float x_imag[1024],
    float y_real[1024],
    float y_imag[1024]
) {
    // Buffers internos para el calculo in-place de la FFT.
    float a_real[FFT_N];
    float a_imag[FFT_N];
    float cos_tab[FFT_N / 2];
    float sin_tab[FFT_N / 2];

    // -------------------------------------------------------------
    // 1) Carga de entrada con permutacion por inversion de bits.
    // -------------------------------------------------------------
BITREV_LOAD:
    for (int i = 0; i < FFT_N; i++) {
#pragma HLS PIPELINE II=1
        int ridx = bit_reverse(i, FFT_LOGN);
        a_real[i] = x_real[ridx];
        a_imag[i] = x_imag[ridx];
    }

    // -------------------------------------------------------------
    // 2) Tabla de factores de giro (twiddle factors) W_N^i,
    //    W_N^i = exp(-j*2*pi*i/N) = cos(theta) - j*sin(theta)
    // -------------------------------------------------------------
TWIDDLE_GEN:
    for (int i = 0; i < FFT_N / 2; i++) {
#pragma HLS PIPELINE II=1
        float angle = -2.0f * PI_F * (float)i / (float)FFT_N;
        cos_tab[i] = cosf(angle);
        sin_tab[i] = sinf(angle);
    }

    // -------------------------------------------------------------
    // 3) Etapas de la FFT iterativa (log2(N) etapas).
    // -------------------------------------------------------------
STAGE_LOOP:
    for (int s = 0; s < FFT_LOGN; s++) {
        int half_m  = 1 << s;
        int m       = half_m << 1;
        int tw_step = FFT_N / m;

    GROUP_LOOP:
        for (int k = 0; k < FFT_N; k += m) {

        BUTTERFLY_LOOP:
            for (int j = 0; j < half_m; j++) {
#pragma HLS PIPELINE II=1
                int idx1   = k + j;
                int idx2   = idx1 + half_m;
                int tw_idx = j * tw_step;

                float ur = a_real[idx1];
                float ui = a_imag[idx1];
                float vr = a_real[idx2];
                float vi = a_imag[idx2];

                float wr = cos_tab[tw_idx];
                float wi = sin_tab[tw_idx];

                // t = W * v
                float tr = wr * vr - wi * vi;
                float ti = wr * vi + wi * vr;

                a_real[idx1] = ur + tr;
                a_imag[idx1] = ui + ti;
                a_real[idx2] = ur - tr;
                a_imag[idx2] = ui - ti;
            }
        }
    }

    // -------------------------------------------------------------
    // 4) Escritura de la salida (ya en orden natural).
    // -------------------------------------------------------------
OUTPUT_STORE:
    for (int i = 0; i < FFT_N; i++) {
#pragma HLS PIPELINE II=1
        y_real[i] = a_real[i];
        y_imag[i] = a_imag[i];
    }
}
