```python
            a0i1: float32 = buf_imag[idx1_1]

            a1i01: float32 = buf_imag[idx2_1]
            a2r01: float32 = buf_real[idx3_1]
            a2i01: float32 = buf_imag[idx3_1]

            a3i01: float32 = buf_imag[idx4_1]
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# Working buffers are accessed with stride patterns that vary per stage
# (1, 4, 16, 64, 256 elements apart) -- any fixed cyclic factor leaves some
# stage's 4 butterfly accesses landing in the same bank (strides that are
# multiples of the factor), so full partitioning is required to guarantee
# conflict-free concurrent access across all stages.
s.partition("kernel:buf_real", partition_type=Partition.Complete)
s.partition("kernel:buf_imag", partition_type=Partition.Complete)

# Digit-reversal permutation and final output copy.
s.pipeline("p", initiation_interval=3)
s.pipeline("c", initiation_interval=3)

# Per-stage butterfly loops: pipeline whichever of (g, j) has more
# iterations for that stage (the one that actually dominates runtime).
# Stage1 (g:256, j:1)   -> g1
# Stage2 (g:64,  j:4)   -> g2
# Stage3 (g:16,  j:16)  -> tie, pipeline g3
# Stage4 (g:4,   j:64)  -> j4
# Stage5 (g:1,   j:256) -> j5
s.pipeline("g1", initiation_interval=3)
s.pipeline("g2", initiation_interval=3)
s.pipeline("g3", initiation_interval=3)
s.pipeline("j4", initiation_interval=3)
s.pipeline("j5", initiation_interval=3)
```
