```python
            idxb10: int32 = idxa10 + 512

            ur10: float32 = buf_real[idxa10]
            ui10: float32 = buf_imag[idxa10]
            vr10: float32 = buf_real[idxb10]
            vi10: float32 = buf_imag[idxb10]

            tr10: float32 = cos_a10 * vr10 - sin_a10 * vi10
            ti10: float32 = cos_a10 * vi10 + sin_a10 * vr10

            buf_real[idxa10] = ur10 + tr10
            buf_imag[idxa10] = ui10 + ti10
            buf_real[idxb10] = ur10 - tr10
            buf_imag[idxb10] = ui10 - ti10

    # Write result to outputs
    for i in range(1024):
        y_real[i] = buf_real[i]
        y_imag[i] = buf_imag[i]
```

```python
s = allo.customize(kernel)

# Buffers are accessed with a per-stage-varying stride (half doubles each
# stage), so Partition.Complete is used instead of a fixed Cyclic/Block

s.partition("kernel:buf_real", partition_type=Partition.Complete)
s.partition("kernel:buf_imag", partition_type=Partition.Complete)

# For each butterfly stage, pipeline whichever of the two nested loops

s.pipeline("g1", initiation_interval=10)
```
