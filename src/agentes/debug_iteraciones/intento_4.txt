### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# buf_real / buf_imag are read/written 4 times per butterfly iteration
# (idxA/idxB/idxC/idxD, stride varying per stage: 1,4,16,64,256) and are
# also touched by the digit-reversal and output-copy loops. objetivo_ii=1
# is the most aggressive target possible, so partition both buffers
# completely to guarantee conflict-free concurrent access regardless of
# stage stride.
s.partition("kernel:buf_real", partition_type=Partition.Complete)
s.partition("kernel:buf_imag", partition_type=Partition.Complete)

# digit-reversal permutation loop
s.pipeline("i0", initiation_interval=1)

# Per stage, pipeline whichever of the two nested loops (g_k / j_k) has
# the larger trip count -- pipelining the trivial one is a no-op and
# leaves the real bottleneck loop unpipelined.
# Stage1: g1=256, j1=1   -> pipeline g1
s.pipeline("g1", initiation_interval=1)
# Stage2: g2=64,  j2=4   -> pipeline g2
s.pipeline("g2", initiation_interval=1)
# Stage3: g3=16,  j3=16  -> tie, pipeline g3
s.pipeline("g3", initiation_interval=1)
# Stage4: g4=4,   j4=64  -> pipeline j4
s.pipeline("j4", initiation_interval=1)
# Stage5: g5=1,   j5=256 -> pipeline j5
s.pipeline("j5", initiation_interval=1)

# output-copy loop
s.pipeline("iout", initiation_interval=1)
```
