### KERNEL
```python
import allo
from allo.ir.types import float32, int32


def kernel(x_real: float32[1024], x_imag: float32[1024],
           y_real: float32[1024], y_imag: float32[1024]):
    # ---- Bit-reversal permutation copy: y[i] = x[bitreverse(i)] ----
    for i in range(1024):
        idx: int32 = i
        rev: int32 = 0
        for b in range(10):
            bit: int32 = idx % 2
            idx = idx / 2
            rev = rev * 2 + bit
        y_real[i] = x_real[rev]
        y_imag[i] = x_imag[rev]

    # ---- Iterative Cooley-Tukey radix-2 DIT butterflies ----
    m: int32 = 1
    for stage in range(10):
        m = m * 2
        half: int32 = m / 2
        num_groups: int32 = 1024 / m
        for g in range(num_groups):
            for j in range(half):
                j_f: float32 = j
                half_f: float32 = half
                angle: float32 = -3.141592653589793 * j_f / half_f

                x2: float32 = angle * angle

                c0: float32 = 1.0
                c1: float32 = -0.5
                c2: float32 = 0.041666666666666664
                c3: float32 = -0.001388888888888889
                c4: float32 = 0.0000248015873015873
                c5: float32 = -0.0000002755731922398589
                c6: float32 = 0.0000000020876756987868
                c7: float32 = -0.0000000000114707455977
                c8: float32 = 0.0000000000000477947733

                cos_val: float32 = c0 + x2 * (c1 + x2 * (c2 + x2 * (c3 + x2 * (c4 + x2 * (c5 + x2 * (c6 + x2 * (c7 + x2 * c8)))))))

                s0: float32 = 1.0
                s1: float32 = -0.16666666666666666
                s2: float32 = 0.008333333333333333
                s3: float32 = -0.0001984126984126984
                s4: float32 = 0.0000027557319223986
                s5: float32 = -0.0000000250521083854
                s6: float32 = 0.0000000001605904384
                s7: float32 = -0.0000000000007647164
                s8: float32 = 0.0000000000000028114

                sin_poly: float32 = s0 + x2 * (s1 + x2 * (s2 + x2 * (s3 + x2 * (s4 + x2 * (s5 + x2 * (s6 + x2 * (s7 + x2 * s8)))))))
                sin_val: float32 = angle * sin_poly

                wr: float32 = cos_val
                wi: float32 = sin_val

                idx_top: int32 = g * m + j
                idx_bot: int32 = idx_top + half

                top_r: float32 = y_real[idx_top]
                top_i: float32 = y_imag[idx_top]
                bot_r: float32 = y_real[idx_bot]
                bot_i: float32 = y_imag[idx_bot]

                tr: float32 = wr * bot_r - wi * bot_i
                ti: float32 = wr * bot_i + wi * bot_r

                y_real[idx_top] = top_r + tr
                y_imag[idx_top] = top_i + ti
                y_real[idx_bot] = top_r - tr
                y_imag[idx_bot] = top_i - ti
```

### SCHEDULE
```python
s = allo.customize(kernel)

from allo.customize import Partition

# El bucle exterior 'stage' tiene solo 10 iteraciones, pero 'half' y
# 'num_groups' (que acotan los bucles internos g/j) dependen de 'stage' y
# no son constantes en tiempo de compilación mientras 'stage' no esté
# desenrollado. Eso es lo que produce II=null y la latencia centinela de
# ~10^14 ciclos: HLS no puede acotar el trip count de g/j. Desenrollar
# 'stage' por completo resuelve half/num_groups como constantes reales en
# cada copia.
s.unroll("stage", factor=0)

# Con 'stage' desenrollado, cada copia accede a y_real/y_imag en las
# posiciones idx_top e idx_bot (dos lecturas + dos escrituras por
# iteración del bucle 'j'). Particionar cíclicamente permite que varias de
# esas iteraciones se solapen sin conflicto de puerto de memoria, lo cual
# es necesario para que el pipeline del bucle 'j' converja a II=2.
s.partition("kernel:y_real", partition_type=Partition.Cyclic, factor=4)
s.partition("kernel:y_imag", partition_type=Partition.Cyclic, factor=4)

# Pipelinea el bucle interno de la mariposa (objetivo_ii=2).
s.pipeline("j", initiation_interval=2)

# Pipelinea el bucle interno de bit-reversal (bound constante = 10),
# eliminando también ahí cualquier riesgo de II indefinido.
s.pipeline("b", initiation_interval=1)

# Pipelinea el bucle exterior de la fase de bit-reversal.
s.pipeline("i", initiation_interval=1)
```
