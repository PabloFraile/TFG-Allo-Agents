### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# Particionado ciclico de los arrays de salida usados en todas las mariposas
# (stride variable segun la etapa) -- factor=8 para objetivo_ii=5.
s.partition("kernel:y_real", partition_type=Partition.Cyclic, factor=8)
s.partition("kernel:y_imag", partition_type=Partition.Cyclic, factor=8)

# Bucle de permutacion bit-reversal
s.pipeline("i", initiation_interval=5)

# Etapas 1-5: el bucle exterior 'g' tiene mas iteraciones que 'j' -> pipelinear 'g'
s.pipeline("g1", initiation_interval=5)
s.pipeline("g2", initiation_interval=5)
s.pipeline("g3", initiation_interval=5)
s.pipeline("g4", initiation_interval=5)
s.pipeline("g5", initiation_interval=5)

# Etapas 6-10: el bucle interior 'j' tiene mas iteraciones que 'g' -> pipelinear 'j'
s.pipeline("j6", initiation_interval=5)
s.pipeline("j7", initiation_interval=5)
s.pipeline("j8", initiation_interval=5)
s.pipeline("j9", initiation_interval=5)
s.pipeline("j10", initiation_interval=5)
```
