```python
            tw_imag1: float32 = 0.0 - sin1
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# Bit-reversal / final copy loops
s.pipeline("n0", initiation_interval=20)
s.pipeline("n1", initiation_interval=20)

# Stages 1-5: the outer group loop 'g' has more iterations than the inner
# twiddle loop 'j', so pipeline 'g'.
s.pipeline("g1", initiation_interval=20)
s.pipeline("g2", initiation_interval=20)
s.pipeline("g3", initiation_interval=20)
s.pipeline("g4", initiation_interval=20)
s.pipeline("g5", initiation_interval=20)

# Stages 6-10: the inner twiddle loop 'j' has more iterations than the outer
# group loop 'g', so pipeline 'j'.
s.pipeline("j6", initiation_interval=20)
s.pipeline("j7", initiation_interval=20)
s.pipeline("j8", initiation_interval=20)
s.pipeline("j9", initiation_interval=20)
s.pipeline("j10", initiation_interval=20)

# Butterfly buffers are accessed with a stride ('half') that changes from
# stage to stage, so a fixed Cyclic/Block factor cannot avoid bank
# conflicts across all stages -> use full partitioning.
s.partition("kernel:buf_real", partition_type=Partition.Complete)
s.partition("kernel:buf_imag", partition_type=Partition.Complete)
```
