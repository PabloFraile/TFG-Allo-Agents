### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

s.partition("kernel:buf_real", partition_type=Partition.Complete)
s.partition("kernel:buf_imag", partition_type=Partition.Complete)

# Pipeline the loop with the most iterations at each stage
# (groups * half = 512 always; pick whichever dominates)
s.pipeline("g1", initiation_interval=10)   # groups=512, half=1
s.pipeline("g2", initiation_interval=10)   # groups=256, half=2
s.pipeline("g3", initiation_interval=10)   # groups=128, half=4
s.pipeline("g4", initiation_interval=10)   # groups=64,  half=8
s.pipeline("g5", initiation_interval=10)   # groups=32,  half=16
s.pipeline("j6", initiation_interval=10)   # groups=16,  half=32
s.pipeline("j7", initiation_interval=10)   # groups=8,   half=64
s.pipeline("j8", initiation_interval=10)   # groups=4,   half=128
s.pipeline("j9", initiation_interval=10)   # groups=2,   half=256
s.pipeline("j10", initiation_interval=10)  # groups=1,   half=512
```
