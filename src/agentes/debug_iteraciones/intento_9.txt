```python
from allo.ir.types import float32, int32


def kernel(x_real: float32[1024], x_imag: float32[1024],
           y_real: float32[1024], y_imag: float32[1024]):
    PI: float32 = 3.14159265358979323846
    TWO_PI: float32 = 6.28318530717958647692

    buf_real: float32[1024]
    buf_imag: float32[1024]

    # ---------------------------------------------------------------
    # Step 0: digit-reversal permutation in base 4 (5 digits, 4^5=1024)
    # ---------------------------------------------------------------
    for i in range(1024):
        d0: int32 = i - (i / 4) * 4
        q1: int32 = i / 4
        d1: int32 = q1 - (q1 / 4) * 4
        q2: int32 = q1 / 4
        d2: int32 = q2 - (q2 / 4) * 4
        q3: int32 = q2 / 4
        d3: int32 = q3 - (q3 / 4) * 4
        d4: int32 = q3 / 4
        ridx: int32 = d0 * 256 + d1 * 64 + d2 * 16 + d3 * 4 + d4
        buf_real[ridx] = x_real[i]
        buf_imag[ridx] = x_imag[i]

    # ===================================================================
    # Stage 1 : m=4, Q=m/4=1, groups G=256  (idx range = G*Q = 256)
    # ===================================================================
    for idx1 in range(256):
        g1: int32 = idx1 / 1
        j1: int32 = idx1 - g1 * 1
        base1: int32 = g1 * 4 + j1

        jf1: float32 = j1
        ang1: float32 = -TWO_PI * jf1 / 4.0
        if ang1 < -PI:
            ang1 = ang1 + TWO_PI

        x2_1: float32 = ang1 * ang1
        p4_1: float32 = x2_1 * x2_1
        p6_1: float32 = p4_1 * x2_1
        p8_1: float32 = p6_1 * x2_1
        p10_1: float32 = p8_1 * x2_1
        p12_1: float32 = p10_1 * x2_1

        cos1_1: float32 = 1.0 - x2_1 / 2.0 + p4_1 / 24.0 - p6_1 / 720.0 + p8_1 / 40320.0 - p10_1 / 3628800.0 + p12_1 / 479001600.0
        sin1_1: float32 = ang1 - ang1 * x2_1 / 6.0 + ang1 * p4_1 / 120.0 - ang1 * p6_1 / 5040.0 + ang1 * p8_1 / 362880.0 - ang1 * p10_1 / 39916800.0 + ang1 * p12_1 / 6227020800.0

        cos2_1: float32 = 2.0 * cos1_1 * cos1_1 - 1.0
        sin2_1: float32 = 2.0 * sin1_1 * cos1_1
        cos3_1: float32 = cos1_1 * (4.0 * cos1_1 * cos1_1 - 3.0)
        sin3_1: float32 = sin1_1 * (3.0 - 4.0 * sin1_1 * sin1_1)

        a0r1: float32 = buf_real[base1]
        a0i1: float32 = buf_imag[base1]
        a1r1: float32 = buf_real[base1 + 1]
        a1i1: float32 = buf_imag[base1 + 1]
        a2r1: float32 = buf_real[base1 + 2]
        a2i1: float32 = buf_imag[base1 + 2]
        a3r1: float32 = buf_real[base1 + 3]
        a3i1: float32 = buf_imag[base1 + 3]

        t1r1: float32 = a1r1 * cos1_1 - a1i1 * sin1_1
        t1i1: float32 = a1r1 * sin1_1 + a1i1 * cos1_1
        t2r1: float32 = a2r1 * cos2_1 - a2i1 * sin2_1
        t2i1: float32 = a2r1 * sin2_1 + a2i1 * cos2_1
        t3r1: float32 = a3r1 * cos3_1 - a3i1 * sin3_1
        t3i1: float32 = a3r1 * sin3_1 + a3i1 * cos3_1

        b0r1: float32 = a0r1 + t1r1 + t2r1 + t3r1
        b0i1: float32 = a0i1 + t1i1 + t2i1 + t3i1
        b1r1: float32 = a0r1 + t1i1 - t2r1 - t3i1
        b1i1: float32 = a0i1 - t1r1 - t2i1 + t3r1
        b2r1: float32 = a0r1 - t1r1 + t2r1 - t3r1
        b2i1: float32 = a0i1 - t1i1 + t2i1 - t3i1
        b3r1: float32 = a0r1 - t1i1 - t2r1 + t3i1
        b3i1: float32 = a0i1 + t1r1 - t2i1 - t3r1

        buf_real[base1] = b0r1
        buf_imag[base1] = b0i1
        buf_real[base1 + 1] = b1r1
        buf_imag[base1 + 1] = b1i1
        buf_real[base1 + 2] = b2r1
        buf_imag[base1 + 2] = b2i1
        buf_real[base1 + 3] = b3r1
        buf_imag[base1 + 3] = b3i1

    # ===================================================================
    # Stage 2 : m=16, Q=4, groups G=64  (idx range = 256)
    # ===================================================================

        sin3_2: float32 = sin1_2 * (3.0 - 4.0 * sin1_2 * sin1_2)

        a0r2: float32 = buf_real[base2]
        a0i2: float32 = buf_imag[base2]
        a1r2: float32 = buf_real[base2 + 4]
        a1i2: float32 = buf_imag[base2 + 4]
        a2r2: float32 = buf_real[base2 + 8]
        a2i2: float32 = buf_imag[base2 + 8]
        a3r2: float32 = buf_real[base2 + 12]
        a3i2: float32 = buf_imag[base2 + 12]

        t1r2: float32 = a1r2 * cos1_2 - a1i2 * sin1_2
        t1i2: float32 = a1r2 * sin1_2 + a1i2 * cos1_2
        t2r2: float32 = a2r2 * cos2_2 - a2i2 * sin2_2
        t2i2: float32 = a2r2 * sin2_2 + a2i2 * cos2_2
        t3r2: float32 = a3r2 * cos3_2 - a3i2 * sin3_2
        t3i2: float32 = a3r2 * sin3_2 + a3i2 * cos3_2

        b0r2: float32 = a0r2 + t1r2 + t2r2 + t3r2
        b0i2: float32 = a0i2 + t1i2 + t2i2 + t3i2
        b1r2: float32 = a0r2 + t1i2 - t2r2 - t3i2
        b1i2: float32 = a0i2 - t1r2 - t2i2 + t3r2
        b2r2: float32 = a0r2 - t1r2 + t2r2 - t3r2
        b2i2: float32 = a0i2 - t1i2 + t2i2 - t3i2
        b3r2: float32 = a0r2 - t1i2 - t2r2 + t3i2
        b3i2: float32 = a0i2 + t1r2 - t2i2 - t3r2

        buf_real[base2] = b0r2
        buf_imag[base2] = b0i2
        buf_real[base2 + 4] = b1r2
        buf_imag[base2 + 4] = b1i2
        buf_real[base2 + 8] = b2r2
        buf_imag[base2 + 8] = b2i2
        buf_real[base2 + 12] = b3r2
        buf_imag[base2 + 12] = b3i2

    # ===================================================================
    # Stage 3 : m=64, Q=16, groups G=16  (idx range = 256)
    # ===================================================================

        g3: int32 = idx3 / 16
        j3: int32 = idx3 - g3 * 16
        base3: int32 = g3 * 64 + j3

        jf3: float32 = j3
        ang3: float32 = -TWO_PI * jf3 / 64.0
        if ang3 < -PI:
            ang3 = ang3 + TWO_PI

        x2_3: float32 = ang3 * ang3
        p4_3: float32 = x2_3 * x2_3
        p6_3: float32 = p4_3 * x2_3
        p8_3: float32 = p6_3 * x2_3
        p10_3: float32 = p8_3 * x2_3
        p12_3: float32 = p10_3 * x2_3

        cos1_3: float32 = 1.0 - x2_3 / 2.0 + p4_3 / 24.0 - p6_3 / 720.0 + p8_3 / 40320.0 - p10_3 / 3628800.0 + p12_3 / 479001600.0
        sin1_3: float32 = ang3 - ang3 * x2_3 / 6.0 + ang3 * p4_3 / 120.0 - ang3 * p6_3 / 5040.0 + ang3 * p8_3 / 362880.0 - ang3 * p10_3 / 39916800.0 + ang3 * p12_3 / 6227020800.0

        cos2_3: float32 = 2.0 * cos1_3 * cos1_3 - 1.0
        sin2_3: float32 = 2.0 * sin1_3 * cos1_3
        cos3_3: float32 = cos1_3 * (4.0 * cos1_3 * cos1_3 - 3.0)
        sin3_3: float32 = sin1_3 * (3.0 - 4.0 * sin1_3 * sin1_3)

        a0r3: float32 = buf_real[base3]
        a0i3: float32 = buf_imag[base3]
        a1r3: float32 = buf_real[base3 + 16]
        a1i3: float32 = buf_imag[base3 + 16]
        a2r3: float32 = buf_real[base3 + 32]
        a2i3: float32 = buf_imag[base3 + 32]
        a3r3: float32 = buf_real[base3 + 48]
        a3i3: float32 = buf_imag[base3 + 48]

        t2r3: float32 = a2r3 * cos2_3 - a2i3 * sin2_3

        t3r3: float32 = a3r3 * cos3_3 - a3i3 * sin3_3
        t3i3: float32 = a3r3 * sin3_3 + a3i3 * cos3_3

        b0r3: float32 = a0r3 + t1r3 + t2r3 + t3r3
        b0i3: float32 = a0i3 + t1i3 + t2i3 + t3i3
        b1r3: float32 = a0r3 + t1i3 - t2r3 - t3i3
        b1i3: float32 = a0i3 - t1r3 - t2i3 + t3r3
        b2r3: float32 = a0r3 - t1r3 + t2r3 - t3r3
        b2i3: float32 = a0i3 - t1i3 + t2i3 - t3i3
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# Moderate cyclic partitioning on the working buffers to provide enough
# concurrent memory ports for the 8 reads + 8 writes per iteration in each
# butterfly stage, without exploding synthesis runtime (avoid Complete).
s.partition("kernel:buf_real", partition_type=Partition.Cyclic, factor=4)
s.partition("kernel:buf_imag", partition_type=Partition.Cyclic, factor=4)

# Pipeline each of the five radix-4 butterfly stages at the requested II.
# Each stage is a single flat loop (idx1..idx5) over 256 iterations, so
# there is no inner/outer ambiguity here -- just pipeline the stage loop.
s.pipeline("idx1", initiation_interval=3)
s.pipeline("idx2", initiation_interval=3)
s.pipeline("idx3", initiation_interval=3)
s.pipeline("idx4", initiation_interval=3)
s.pipeline("idx5", initiation_interval=3)

# Lightweight pipelining of the simple copy loops (digit-reversal load and
# final store) -- cheap, no extra partitioning/unrolling needed.
s.pipeline("i", initiation_interval=3)
s.pipeline("k", initiation_interval=3)
```
