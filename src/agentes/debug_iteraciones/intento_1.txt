```python
            y3_r: float32 = a_r - b_i - c_r + d_i
            y3_i: float32 = a_i + b_r - c_i - d_r

            buf_real[idx0] = y0_r
            buf_imag[idx0] = y0_i
            buf_real[idx1] = y1_r
            buf_imag[idx1] = y1_i
            buf_real[idx2] = y2_r
            buf_imag[idx2] = y2_i
            buf_real[idx3] = y3_r
            buf_imag[idx3] = y3_i

    # ================= Stage 4: M=1024, Q=256, G=1 =================
    af4: float32 = -0.006135923151542565
    for g4 in range(1):
        base4: int32 = g4 * 1024
        for k4 in range(256):
            idx0: int32 = base4 + k4
            idx1: int32 = idx0 + 256
            idx2: int32 = idx1 + 256
            idx3: int32 = idx2 + 256

            a_r: float32 = buf_real[idx0]
            a_i: float32 = buf_imag[idx0]
            b_r0: float32 = buf_real[idx1]
            b_i0: float32 = buf_imag[idx1]
            c_r0: float32 = buf_real[idx2]
            c_i0: float32 = buf_imag[idx2]
            d_r0: float32 = buf_real[idx3]
            d_i0: float32 = buf_imag[idx3]

            kp1: int32 = k4 * 1
            shift1: int32 = (kp1 + 512) / 1024
            wkp1: int32 = kp1 - shift1 * 1024
            wkp1_f: float32 = wkp1
            x1: float32 = af4 * wkp1_f
            x1_2: float32 = x1 * x1
            cos1: float32 = 1.0 - x1_2 / 2.0 + (x1_2 * x1_2) / 24.0 - (x1_2 * x1_2 * x1_2) / 720.0 + (x1_2 * x1_2 * x1_2 * x1_2) / 40320.0 - (x1_2 * x1_2 * x1_2 * x1_2 * x1_2) / 3628800.0 + (x1_2 * x1_2 * x1_2 * x1_2 * x1_2 * x1_2) / 479001600.0
            sin1: float32 = x1 - x1 * x1_2 / 6.0 + x1 * (x1_2 * x1_2) / 120.0 - x1 * (x1_2 * x1_2 * x1_2) / 5040.0 + x1 * (x1_2 * x1_2 * x1_2 * x1_2) / 362880.0 - x1 * (x1_2 * x1_2 * x1_2 * x1_2 * x1_2) / 39916800.0
            b_r: float32 = b_r0 * cos1 - b_i0 * sin1
            b_i: float32 = b_r0 * sin1 + b_i0 * cos1

            kp2: int32 = k4 * 2
            shift2: int32 = (kp2 + 512) / 1024
            wkp2: int32 = kp2 - shift2 * 1024
            wkp2_f: float32 = wkp2
            x2v: float32 = af4 * wkp2_f
            x2v_2: float32 = x2v * x2v
            cos2: float32 = 1.0 - x2v_2 / 2.0 + (x2v_2 * x2v_2) / 24.0 - (x2v_2 * x2v_2 * x2v_2) / 720.0 + (x2v_2 * x2v_2 * x2v_2 * x2v_2) / 40320.0 - (x2v_2 * x2v_2 * x2v_2 * x2v_2 * x2v_2) / 3628800.0 + (x2v_2 * x2v_2 * x2v_2 * x2v_2 * x2v_2 * x2v_2) / 479001600.0
            sin2: float32 = x2v - x2v * x2v_2 / 6.0 + x2v * (x2v_2 * x2v_2) / 120.0 - x2v * (x2v_2 * x2v_2 * x2v_2) / 5040.0 + x2v * (x2v_2 * x2v_2 * x2v_2 * x2v_2) / 362880.0 - x2v * (x2v_2 * x2v_2 * x2v_2 * x2v_2 * x2v_2) / 39916800.0
            c_r: float32 = c_r0 * cos2 - c_i0 * sin2
            c_i: float32 = c_r0 * sin2 + c_i0 * cos2

            kp3: int32 = k4 * 3
            shift3: int32 = (kp3 + 512) / 1024
            wkp3: int32 = kp3 - shift3 * 1024
            wkp3_f: float32 = wkp3
            x3v: float32 = af4 * wkp3_f
            x3v_2: float32 = x3v * x3v
            cos3: float32 = 1.0 - x3v_2 / 2.0 + (x3v_2 * x3v_2) / 24.0 - (x3v_2 * x3v_2 * x3v_2) / 720.0 + (x3v_2 * x3v_2 * x3v_2 * x3v_2) / 40320.0 - (x3v_2 * x3v_2 * x3v_2 * x3v_2 * x3v_2) / 3628800.0 + (x3v_2 * x3v_2 * x3v_2 * x3v_2 * x3v_2 * x3v_2) / 479001600.0
            sin3: float32 = x3v - x3v * x3v_2 / 6.0 + x3v * (x3v_2 * x3v_2) / 120.0 - x3v * (x3v_2 * x3v_2 * x3v_2) / 5040.0 + x3v * (x3v_2 * x3v_2 * x3v_2 * x3v_2) / 362880.0 - x3v * (x3v_2 * x3v_2 * x3v_2 * x3v_2 * x3v_2) / 39916800.0
            d_r: float32 = d_r0 * cos3 - d_i0 * sin3
            d_i: float32 = d_r0 * sin3 + d_i0 * cos3

            y0_r: float32 = a_r + b_r + c_r + d_r
            y0_i: float32 = a_i + b_i + c_i + d_i
            y1_r: float32 = a_r + b_i - c_r - d_i
            y1_i: float32 = a_i - b_r - c_i + d_r
            y2_r: float32 = a_r - b_r + c_r - d_r
            y2_i: float32 = a_i - b_i + c_i - d_i
            y3_r: float32 = a_r - b_i - c_r + d_i
            y3_i: float32 = a_i + b_r - c_i - d_r

            buf_real[idx0] = y0_r
            buf_imag[idx0] = y0_i
            buf_real[idx1] = y1_r
            buf_imag[idx1] = y1_i
            buf_real[idx2] = y2_r
            buf_imag[idx2] = y2_i
            buf_real[idx3] = y3_r
            buf_imag[idx3] = y3_i

    # ---------------- Write results to outputs ----------------
    for i in range(1024):
        y_real[i] = buf_real[i]
        y_imag[i] = buf_imag[i]
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# Strides between the four butterfly taps are powers of 4 (1,4,16,64,256),
# so any fixed cyclic factor that divides a stage's stride collides for
# that stage; a full partition is required to keep every stage conflict-free.
s.partition("kernel:buf_real", partition_type=Partition.Complete)
s.partition("kernel:buf_imag", partition_type=Partition.Complete)

# Pipeline whichever loop in each stage's (g, k) nest has the most
# iterations (the one that actually dominates cycle count).
s.pipeline("g0", initiation_interval=3)   # stage0: g0:256, k0:1
s.pipeline("g1", initiation_interval=3)   # stage1: g1:64,  k1:4
s.pipeline("k2", initiation_interval=3)   # stage2: g2:16,  k2:16 (tie)
s.pipeline("k3", initiation_interval=3)   # stage3: g3:4,   k3:64
s.pipeline("k4", initiation_interval=3)   # stage4: g4:1,   k4:256
```
