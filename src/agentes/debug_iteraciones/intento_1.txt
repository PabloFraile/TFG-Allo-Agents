### KERNEL
```python
import allo
from allo.ir.types import float32, int32

def kernel(x_real: float32[1024], x_imag: float32[1024], y_real: float32[1024], y_imag: float32[1024]):
    buf_real: float32[1024]
    buf_imag: float32[1024]

    # Permutacion digit-reversal base-4 (5 digitos) hacia buf
    for ip in range(1024):
        n0: int32 = ip
        d0: int32 = n0 % 4
        n1: int32 = n0 / 4
        d1: int32 = n1 % 4
        n2: int32 = n1 / 4
        d2: int32 = n2 % 4
        n3: int32 = n2 / 4
        d3: int32 = n3 % 4
        n4: int32 = n3 / 4
        d4: int32 = n4 % 4
        ri: int32 = d0 * 256 + d1 * 64 + d2 * 16 + d3 * 4 + d4
        buf_real[ip] = x_real[ri]
        buf_imag[ip] = x_imag[ri]

    # Etapa 0: m=4, m4=1, num_grupos=256 (twiddles triviales, w=1)
    for g0 in range(256):
        i0_0: int32 = g0 * 4
        i1_0: int32 = i0_0 + 1
        i2_0: int32 = i0_0 + 2
        i3_0: int32 = i0_0 + 3

        a0r_0: float32 = buf_real[i0_0]
        a0i_0: float32 = buf_imag[i0_0]
        a1r_0: float32 = buf_real[i1_0]
        a1i_0: float32 = buf_imag[i1_0]
        a2r_0: float32 = buf_real[i2_0]
        a2i_0: float32 = buf_imag[i2_0]
        a3r_0: float32 = buf_real[i3_0]
        a3i_0: float32 = buf_imag[i3_0]

        t0r_0: float32 = a0r_0 + a2r_0
        t0i_0: float32 = a0i_0 + a2i_0
        t1r_0: float32 = a0r_0 - a2r_0
        t1i_0: float32 = a0i_0 - a2i_0
        t2r_0: float32 = a1r_0 + a3r_0
        t2i_0: float32 = a1i_0 + a3i_0
        t3r_0: float32 = a1r_0 - a3r_0
        t3i_0: float32 = a1i_0 - a3i_0

        buf_real[i0_0] = t0r_0 + t2r_0
        buf_imag[i0_0] = t0i_0 + t2i_0
        buf_real[i2_0] = t0r_0 - t2r_0
        buf_imag[i2_0] = t0i_0 - t2i_0
        buf_real[i1_0] = t1r_0 + t3i_0
        buf_imag[i1_0] = t1i_0 - t3r_0
        buf_real[i3_0] = t1r_0 - t3i_0
        buf_imag[i3_0] = t1i_0 + t3r_0

    # Etapa 1: m=16, m4=4, num_grupos=64, coef = -2*pi/16
    for g1 in range(64):
        for j1 in range(4):
            j_f_1: float32 = j1
            theta1_1: float32 = -0.39269908169872414 * j_f_1

            x2_1: float32 = theta1_1 * theta1_1
            x3_1: float32 = x2_1 * theta1_1
            x4_1: float32 = x2_1 * x2_1
            x5_1: float32 = x4_1 * theta1_1
            x6_1: float32 = x4_1 * x2_1
            x7_1: float32 = x6_1 * theta1_1
            x8_1: float32 = x4_1 * x4_1
            x9_1: float32 = x8_1 * theta1_1
            x10_1: float32 = x8_1 * x2_1
            x11_1: float32 = x10_1 * theta1_1

            cos1_1: float32 = 1.0 - x2_1 / 2.0 + x4_1 / 24.0 - x6_1 / 720.0 + x8_1 / 40320.0 - x10_1 / 3628800.0
            sin1_1: float32 = theta1_1 - x3_1 / 6.0 + x5_1 / 120.0 - x7_1 / 5040.0 + x9_1 / 362880.0 - x11_1 / 39916800.0

            cos2_1: float32 = cos1_1 * cos1_1 - sin1_1 * sin1_1
            sin2_1: float32 = 2.0 * cos1_1 * sin1_1
            cos3_1: float32 = cos2_1 * cos1_1 - sin2_1 * sin1_1
            sin3_1: float32 = cos2_1 * sin1_1 + sin2_1 * cos1_1

            i0_1: int32 = g1 * 16 + j1
            i1_1: int32 = i0_1 + 4
            i2_1: int32 = i1_1 + 4
            i3_1: int32 = i2_1 + 4

            a0r_1: float32 = buf_real[i0_1]
            a0i_1: float32 = buf_imag[i0_1]
            a1r_1: float32 = buf_real[i1_1]
            a1i_1: float32 = buf_imag[i1_1]
            a2r_1: float32 = buf_real[i2_1]
            a2i_1: float32 = buf_imag[i2_1]
            a3r_1: float32 = buf_real[i3_1]
            a3i_1: float32 = buf_imag[i3_1]

            Br_1: float32 = a1r_1 * cos1_1 - a1i_1 * sin1_1
            Bi_1: float32 = a1r_1 * sin1_1 + a1i_1 * cos1_1
            Cr_1: float32 = a2r_1 * cos2_1 - a2i_1 * sin2_1
            Ci_1: float32 = a2r_1 * sin2_1 + a2i_1 * cos2_1
            Dr_1: float32 = a3r_1 * cos3_1 - a3i_1 * sin3_1
            Di_1: float32 = a3r_1 * sin3_1 + a3i_1 * cos3_1

            t0r_1: float32 = a0r_1 + Cr_1
            t0i_1: float32 = a0i_1 + Ci_1
            t1r_1: float32 = a0r_1 - Cr_1
            t1i_1: float32 = a0i_1 - Ci_1
            t2r_1: float32 = Br_1 + Dr_1
            t2i_1: float32 = Bi_1 + Di_1
            t3r_1: float32 = Br_1 - Dr_1
            t3i_1: float32 = Bi_1 - Di_1

            buf_real[i0_1] = t0r_1 + t2r_1
            buf_imag[i0_1] = t0i_1 + t2i_1
            buf_real[i2_1] = t0r_1 - t2r_1
            buf_imag[i2_1] = t0i_1 - t2i_1
            buf_real[i1_1] = t1r_1 + t3i_1
            buf_imag[i1_1] = t1i_1 - t3r_1
            buf_real[i3_1] = t1r_1 - t3i_1
            buf_imag[i3_1] = t1i_1 + t3r_1

    # Etapa 2: m=64, m4=16, num_grupos=16, coef = -2*pi/64
    for g2 in range(16):
        for j2 in range(16):
            j_f_2: float32 = j2
            theta1_2: float32 = -0.09817477042468103 * j_f_2

            x2_2: float32 = theta1_2 * theta1_2
            x3_2: float32 = x2_2 * theta1_2
            x4_2: float32 = x2_2 * x2_2
            x5_2: float32 = x4_2 * theta1_2
            x6_2: float32 = x4_2 * x2_2
            x7_2: float32 = x6_2 * theta1_2
            x8_2: float32 = x4_2 * x4_2
            x9_2: float32 = x8_2 * theta1_2
            x10_2: float32 = x8_2 * x2_2
            x11_2: float32 = x10_2 * theta1_2

            cos1_2: float32 = 1.0 - x2_2 / 2.0 + x4_2 / 24.0 - x6_2 / 720.0 + x8_2 / 40320.0 - x10_2 / 3628800.0
            sin1_2: float32 = theta1_2 - x3_2 / 6.0 + x5_2 / 120.0 - x7_2 / 5040.0 + x9_2 / 362880.0 - x11_2 / 39916800.0

            cos2_2: float32 = cos1_2 * cos1_2 - sin1_2 * sin1_2
            sin2_2: float32 = 2.0 * cos1_2 * sin1_2
            cos3_2: float32 = cos2_2 * cos1_2 - sin2_2 * sin1_2
            sin3_2: float32 = cos2_2 * sin1_2 + sin2_2 * cos1_2

            i0_2: int32 = g2 * 64 + j2
            i1_2: int32 = i0_2 + 16
            i2_2: int32 = i1_2 + 16
            i3_2: int32 = i2_2 + 16

            a0r_2: float32 = buf_real[i0_2]
            a0i_2: float32 = buf_imag[i0_2]
            a1r_2: float32 = buf_real[i1_2]
            a1i_2: float32 = buf_imag[i1_2]
            a2r_2: float32 = buf_real[i2_2]
            a2i_2: float32 = buf_imag[i2_2]
            a3r_2: float32 = buf_real[i3_2]
            a3i_2: float32 = buf_imag[i3_2]

            Br_2: float32 = a1r_2 * cos1_2 - a1i_2 * sin1_2
            Bi_2: float32 = a1r_2 * sin1_2 + a1i_2 * cos1_2
            Cr_2: float32 = a2r_2 * cos2_2 - a2i_2 * sin2_2
            Ci_2: float32 = a2r_2 * sin2_2 + a2i_2 * cos2_2
            Dr_2: float32 = a3r_2 * cos3_2 - a3i_2 * sin3_2
            Di_2: float32 = a3r_2 * sin3_2 + a3i_2 * cos3_2

            t0r_2: float32 = a0r_2 + Cr_2
            t0i_2: float32 = a0i_2 + Ci_2
            t1r_2: float32 = a0r_2 - Cr_2
            t1i_2: float32 = a0i_2 - Ci_2
            t2r_2: float32 = Br_2 + Dr_2
            t2i_2: float32 = Bi_2 + Di_2
            t3r_2: float32 = Br_2 - Dr_2
            t3i_2: float32 = Bi_2 - Di_2

            buf_real[i0_2] = t0r_2 + t2r_2
            buf_imag[i0_2] = t0i_2 + t2i_2
            buf_real[i2_2] = t0r_2 - t2r_2
            buf_imag[i2_2] = t0i_2 - t2i_2
            buf_real[i1_2] = t1r_2 + t3i_2
            buf_imag[i1_2] = t1i_2 - t3r_2
            buf_real[i3_2] = t1r_2 - t3i_2
            buf_imag[i3_2] = t1i_2 + t3r_2

    # Etapa 3: m=256, m4=64, num_grupos=4, coef = -2*pi/256
    for g3 in range(4):
        for j3 in range(64):
            j_f_3: float32 = j3
            theta1_3: float32 = -0.024543692606170259 * j_f_3

            x2_3: float32 = theta1_3 * theta1_3
            x3_3: float32 = x2_3 * theta1_3
            x4_3: float32 = x2_3 * x2_3
            x5_3: float32 = x4_3 * theta1_3
            x6_3: float32 = x4_3 * x2_3
            x7_3: float32 = x6_3 * theta1_3
            x8_3: float32 = x4_3 * x4_3
            x9_3: float32 = x8_3 * theta1_3
            x10_3: float32 = x8_3 * x2_3
            x11_3: float32 = x10_3 * theta1_3

            cos1_3: float32 = 1.0 - x2_3 / 2.0 + x4_3 / 24.0 - x6_3 / 720.0 + x8_3 / 40320.0 - x10_3 / 3628800.0
            sin1_3: float32 = theta1_3 - x3_3 / 6.0 + x5_3 / 120.0 - x7_3 / 5040.0 + x9_3 / 362880.0 - x11_3 / 39916800.0

            cos2_3: float32 = cos1_3 * cos1_3 - sin1_3 * sin1_3
            sin2_3: float32 = 2.0 * cos1_3 * sin1_3
            cos3_3: float32 = cos2_3 * cos1_3 - sin2_3 * sin1_3
            sin3_3: float32 = cos2_3 * sin1_3 + sin2_3 * cos1_3

            i0_3: int32 = g3 * 256 + j3
            i1_3: int32 = i0_3 + 64
            i2_3: int32 = i1_3 + 64
            i3_3: int32 = i2_3 + 64

            a0r_3: float32 = buf_real[i0_3]
            a0i_3: float32 = buf_imag[i0_3]
            a1r_3: float32 = buf_real[i1_3]
            a1i_3: float32 = buf_imag[i1_3]
            a2r_3: float32 = buf_real[i2_3]
            a2i_3: float32 = buf_imag[i2_3]
            a3r_3: float32 = buf_real[i3_3]
            a3i_3: float32 = buf_imag[i3_3]

            Br_3: float32 = a1r_3 * cos1_3 - a1i_3 * sin1_3
            Bi_3: float32 = a1r_3 * sin1_3 + a1i_3 * cos1_3
            Cr_3: float32 = a2r_3 * cos2_3 - a2i_3 * sin2_3
            Ci_3: float32 = a2r_3 * sin2_3 + a2i_3 * cos2_3
            Dr_3: float32 = a3r_3 * cos3_3 - a3i_3 * sin3_3
            Di_3: float32 = a3r_3 * sin3_3 + a3i_3 * cos3_3

            t0r_3: float32 = a0r_3 + Cr_3
            t0i_3: float32 = a0i_3 + Ci_3
            t1r_3: float32 = a0r_3 - Cr_3
            t1i_3: float32 = a0i_3 - Ci_3
            t2r_3: float32 = Br_3 + Dr_3
            t2i_3: float32 = Bi_3 + Di_3
            t3r_3: float32 = Br_3 - Dr_3
            t3i_3: float32 = Bi_3 - Di_3

            buf_real[i0_3] = t0r_3 + t2r_3
            buf_imag[i0_3] = t0i_3 + t2i_3
            buf_real[i2_3] = t0r_3 - t2r_3
            buf_imag[i2_3] = t0i_3 - t2i_3
            buf_real[i1_3] = t1r_3 + t3i_3
            buf_imag[i1_3] = t1i_3 - t3r_3
            buf_real[i3_3] = t1r_3 - t3i_3
            buf_imag[i3_3] = t1i_3 + t3r_3

    # Etapa 4: m=1024, m4=256, num_grupos=1 (g4=0 constante), coef = -2*pi/1024
    for j4 in range(256):
        j_f_4: float32 = j4
        theta1_4: float32 = -0.006135923151542565 * j_f_4

        x2_4: float32 = theta1_4 * theta1_4
        x3_4: float32 = x2_4 * theta1_4
        x4_4: float32 = x2_4 * x2_4
        x5_4: float32 = x4_4 * theta1_4
        x6_4: float32 = x4_4 * x2_4
        x7_4: float32 = x6_4 * theta1_4
        x8_4: float32 = x4_4 * x4_4
        x9_4: float32 = x8_4 * theta1_4
        x10_4: float32 = x8_4 * x2_4
        x11_4: float32 = x10_4 * theta1_4

        cos1_4: float32 = 1.0 - x2_4 / 2.0 + x4_4 / 24.0 - x6_4 / 720.0 + x8_4 / 40320.0 - x10_4 / 3628800.0
        sin1_4: float32 = theta1_4 - x3_4 / 6.0 + x5_4 / 120.0 - x7_4 / 5040.0 + x9_4 / 362880.0 - x11_4 / 39916800.0

        cos2_4: float32 = cos1_4 * cos1_4 - sin1_4 * sin1_4
        sin2_4: float32 = 2.0 * cos1_4 * sin1_4
        cos3_4: float32 = cos2_4 * cos1_4 - sin2_4 * sin1_4
        sin3_4: float32 = cos2_4 * sin1_4 + sin2_4 * cos1_4

        i0_4: int32 = j4
        i1_4: int32 = i0_4 + 256
        i2_4: int32 = i1_4 + 256
        i3_4: int32 = i2_4 + 256

        a0r_4: float32 = buf_real[i0_4]
        a0i_4: float32 = buf_imag[i0_4]
        a1r_4: float32 = buf_real[i1_4]
        a1i_4: float32 = buf_imag[i1_4]
        a2r_4: float32 = buf_real[i2_4]
        a2i_4: float32 = buf_imag[i2_4]
        a3r_4: float32 = buf_real[i3_4]
        a3i_4: float32 = buf_imag[i3_4]

        Br_4: float32 = a1r_4 * cos1_4 - a1i_4 * sin1_4
        Bi_4: float32 = a1r_4 * sin1_4 + a1i_4 * cos1_4
        Cr_4: float32 = a2r_4 * cos2_4 - a2i_4 * sin2_4
        Ci_4: float32 = a2r_4 * sin2_4 + a2i_4 * cos2_4
        Dr_4: float32 = a3r_4 * cos3_4 - a3i_4 * sin3_4
        Di_4: float32 = a3r_4 * sin3_4 + a3i_4 * cos3_4

        t0r_4: float32 = a0r_4 + Cr_4
        t0i_4: float32 = a0i_4 + Ci_4
        t1r_4: float32 = a0r_4 - Cr_4
        t1i_4: float32 = a0i_4 - Ci_4
        t2r_4: float32 = Br_4 + Dr_4
        t2i_4: float32 = Bi_4 + Di_4
        t3r_4: float32 = Br_4 - Dr_4
        t3i_4: float32 = Bi_4 - Di_4

        buf_real[i0_4] = t0r_4 + t2r_4

        buf_real[i2_4] = t0r_4 - t2r_4
        buf_imag[i2_4] = t0i_4 - t2i_4
        buf_real[i1_4] = t1r_4 + t3i_4
        buf_imag[i1_4] = t1i_4 - t3r_4
        buf_real[i3_4] = t1r_4 - t3i_4

    # Copia del resultado final (orden natural tras la permutacion digit-reversed inicial)
    for ic in range(1024):
        y_real[ic] = buf_real[ic]
        y_imag[ic] = buf_imag[ic]
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

s.partition("kernel:buf_real", partition_type=Partition.Cyclic, factor=16)
s.partition("kernel:buf_imag", partition_type=Partition.Cyclic, factor=16)

s.pipeline("ip", initiation_interval=5)

# Etapa 0: g0 (256 iter) domina sobre el twiddle trivial -> pipelinear g0
s.pipeline("g0", initiation_interval=5)

# Etapa 1: g1 (64 iter) domina sobre j1 (4 iter) -> pipelinear g1
s.pipeline("g1", initiation_interval=5)

# Etapa 2: g2 y j2 empatan (16 iter cada uno) -> pipelinear g2
s.pipeline("g2", initiation_interval=5)

# Etapa 3: j3 (64 iter) domina sobre g3 (4 iter) -> pipelinear j3
s.pipeline("j3", initiation_interval=5)

# Etapa 4: unico bucle real, j4 (256 iter) -> pipelinear j4
s.pipeline("j4", initiation_interval=5)

s.pipeline("ic", initiation_interval=5)
```
