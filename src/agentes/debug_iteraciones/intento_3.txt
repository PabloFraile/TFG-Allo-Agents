```python
import allo
from allo.ir.types import float32, int32


def kernel(x_real: float32[1024], x_imag: float32[1024],
           y_real: float32[1024], y_imag: float32[1024]):

    # ---- Tabla de twiddle factors (512 entradas), cada una calculada de
    # forma independiente a partir de su indice k (sin recurrencia) usando
    # una serie de Taylor de seno/coseno ----
    twiddle_real: float32[512]
    twiddle_imag: float32[512]

    NEG_TWO_PI: float32 = -6.283185307179586

    for k in range(512):
        k_f: float32 = k
        ang: float32 = NEG_TWO_PI * k_f / 1024.0

        x2: float32 = ang * ang
        x3: float32 = x2 * ang
        x4: float32 = x2 * x2
        x5: float32 = x4 * ang
        x6: float32 = x4 * x2
        x7: float32 = x6 * ang
        x8: float32 = x4 * x4
        x9: float32 = x8 * ang
        x10: float32 = x8 * x2
        x11: float32 = x10 * ang
        x12: float32 = x8 * x4
        x13: float32 = x12 * ang
        x14: float32 = x8 * x6
        x15: float32 = x14 * ang
        x16: float32 = x8 * x8

        cos_val: float32 = 1.0 - x2 / 2.0 + x4 / 24.0 - x6 / 720.0 + x8 / 40320.0 - x10 / 3628800.0 + x12 / 479001600.0 - x14 / 87178291200.0 + x16 / 20922789888000.0
        sin_val: float32 = ang - x3 / 6.0 + x5 / 120.0 - x7 / 5040.0 + x9 / 362880.0 - x11 / 39916800.0 + x13 / 6227020800.0 - x15 / 1307674368000.0

        twiddle_real[k] = cos_val
        twiddle_imag[k] = sin_val

    # ---- Permutacion bit-reversal (10 bits, N=1024), calculada dentro del
    # kernel sin modulo (solo +,-,*,/) ----
    for i in range(1024):
        idx: int32 = i
        rev: int32 = 0
        for b in range(10):
            half_idx: int32 = idx / 2
            bit: int32 = idx - half_idx * 2
            idx = half_idx
            rev = rev * 2 + bit
        y_real[rev] = x_real[i]
        y_imag[rev] = x_imag[i]

    # ---- Etapa 1: m=2, half=1, num_groups=512, stride=512 ----
    for g1 in range(512):
        base_1: int32 = g1 * 2
        for j1 in range(1):
            idx1_1: int32 = base_1 + j1
            idx2_1: int32 = idx1_1 + 1
            tw_idx_1: int32 = j1 * 512
            tr_1: float32 = twiddle_real[tw_idx_1]
            ti_1: float32 = twiddle_imag[tw_idx_1]
            ur_1: float32 = y_real[idx1_1]
            ui_1: float32 = y_imag[idx1_1]
            vr_1: float32 = y_real[idx2_1]
            vi_1: float32 = y_imag[idx2_1]
            trv_1: float32 = tr_1 * vr_1 - ti_1 * vi_1
            tiv_1: float32 = tr_1 * vi_1 + ti_1 * vr_1
            y_real[idx1_1] = ur_1 + trv_1
            y_imag[idx1_1] = ui_1 + tiv_1
            y_real[idx2_1] = ur_1 - trv_1
            y_imag[idx2_1] = ui_1 - tiv_1

    # ---- Etapa 2: m=4, half=2, num_groups=256, stride=256 ----
    for g2 in range(256):
        base_2: int32 = g2 * 4
        for j2 in range(2):
            idx1_2: int32 = base_2 + j2
            idx2_2: int32 = idx1_2 + 2
            tw_idx_2: int32 = j2 * 256
            tr_2: float32 = twiddle_real[tw_idx_2]
            ti_2: float32 = twiddle_imag[tw_idx_2]
            ur_2: float32 = y_real[idx1_2]
            ui_2: float32 = y_imag[idx1_2]
            vr_2: float32 = y_real[idx2_2]
            vi_2: float32 = y_imag[idx2_2]
            trv_2: float32 = tr_2 * vr_2 - ti_2 * vi_2
            tiv_2: float32 = tr_2 * vi_2 + ti_2 * vr_2
            y_real[idx1_2] = ur_2 + trv_2
            y_imag[idx1_2] = ui_2 + tiv_2
            y_real[idx2_2] = ur_2 - trv_2
            y_imag[idx2_2] = ui_2 - tiv_2

    # ---- Etapa 3: m=8, half=4, num_groups=128, stride=128 ----
    for g3 in range(128):
        base_3: int32 = g3 * 8
        for j3 in range(4):
            idx1_3: int32 = base_3 + j3
            idx2_3: int32 = idx1_3 + 4
            tw_idx_3: int32 = j3 * 128
            tr_3: float32 = twiddle_real[tw_idx_3]
            ti_3: float32 = twiddle_imag[tw_idx_3]
            ur_3: float32 = y_real[idx1_3]
            ui_3: float32 = y_imag[idx1_3]
            vr_3: float32 = y_real[idx2_3]
            vi_3: float32 = y_imag[idx2_3]
            trv_3: float32 = tr_3 * vr_3 - ti_3 * vi_3
            tiv_3: float32 = tr_3 * vi_3 + ti_3 * vr_3
            y_real[idx1_3] = ur_3 + trv_3
            y_imag[idx1_3] = ui_3 + tiv_3
            y_real[idx2_3] = ur_3 - trv_3
            y_imag[idx2_3] = ui_3 - tiv_3

    # ---- Etapa 4: m=16, half=8, num_groups=64, stride=64 ----
    for g4 in range(64):
        base_4: int32 = g4 * 16
        for j4 in range(8):
            idx1_4: int32 = base_4 + j4
            idx2_4: int32 = idx1_4 + 8
            tw_idx_4: int32 = j4 * 64
            tr_4: float32 = twiddle_real[tw_idx_4]
            ti_4: float32 = twiddle_imag[tw_idx_4]
            ur_4: float32 = y_real[idx1_4]
            ui_4: float32 = y_imag[idx1_4]
            vr_4: float32 = y_real[idx2_4]
            vi_4: float32 = y_imag[idx2_4]
            trv_4: float32 = tr_4 * vr_4 - ti_4 * vi_4
            tiv_4: float32 = tr_4 * vi_4 + ti_4 * vr_4
            y_real[idx1_4] = ur_4 + trv_4
            y_imag[idx1_4] = ui_4 + tiv_4
            y_real[idx2_4] = ur_4 - trv_4
            y_imag[idx2_4] = ui_4 - tiv_4

    # ---- Etapa 5: m=32, half=16, num_groups=32, stride=32 ----
    for g5 in range(32):
        base_5: int32 = g5 * 32
        for j5 in range(16):
            idx1_5: int32 = base_5 + j5
            idx2_5: int32 = idx1_5 + 16
            tw_idx_5: int32 = j5 * 32
            tr_5: float32 = twiddle_real[tw_idx_5]
            ti_5: float32 = twiddle_imag[tw_idx_5]
            ur_5: float32 = y_real[idx1_5]
            ui_5: float32 = y_imag[idx1_5]
            vr_5: float32 = y_real[idx2_5]
            vi_5: float32 = y_imag[idx2_5]
            trv_5: float32 = tr_5 * vr_5 - ti_5 * vi_5
            tiv_5: float32 = tr_5 * vi_5 + ti_5 * vr_5
            y_real[idx1_5] = ur_5 + trv_5
            y_imag[idx1_5] = ui_5 + tiv_5
            y_real[idx2_5] = ur_5 - trv_5
            y_imag[idx2_5] = ui_5 - tiv_5

    # ---- Etapa 6: m=64, half=32, num_groups=16, stride=16 ----
    for g6 in range(16):
        base_6: int32 = g6 * 64
        for j6 in range(32):
            idx1_6: int32 = base_6 + j6
            idx2_6: int32 = idx1_6 + 32
            tw_idx_6: int32 = j6 * 16
            tr_6: float32 = twiddle_real[tw_idx_6]
            ti_6: float32 = twiddle_imag[tw_idx_6]
            ur_6: float32 = y_real[idx1_6]
            ui_6: float32 = y_imag[idx1_6]
            vr_6: float32 = y_real[idx2_6]
            vi_6: float32 = y_imag[idx2_6]
            trv_6: float32 = tr_6 * vr_6 - ti_6 * vi_6
            tiv_6: float32 = tr_6 * vi_6 + ti_6 * vr_6
            y_real[idx1_6] = ur_6 + trv_6
            y_imag[idx1_6] = ui_6 + tiv_6
            y_real[idx2_6] = ur_6 - trv_6
            y_imag[idx2_6] = ui_6 - tiv_6

    # ---- Etapa 7: m=128, half=64, num_groups=8, stride=8 ----
    for g7 in range(8):
        base_7: int32 = g7 * 128
        for j7 in range(64):
            idx1_7: int32 = base_7 + j7
            idx2_7: int32 = idx1_7 + 64
            tw_idx_7: int32 = j7 * 8
            tr_7: float32 = twiddle_real[tw_idx_7]
            ti_7: float32 = twiddle_imag[tw_idx_7]
            ur_7: float32 = y_real[idx1_7]
            ui_7: float32 = y_imag[idx1_7]
            vr_7: float32 = y_real[idx2_7]
            vi_7: float32 = y_imag[idx2_7]
            trv_7: float32 = tr_7 * vr_7 - ti_7 * vi_7
            tiv_7: float32 = tr_7 * vi_7 + ti_7 * vr_7
            y_real[idx1_7] = ur_7 + trv_7
            y_imag[idx1_7] = ui_7 + tiv_7
            y_real[idx2_7] = ur_7 - trv_7

    # ---- Etapa 8: m=256, half=128, num_groups=4, stride=4 ----
    for g8 in range(4):
        base_8: int32 = g8 * 256
        for j8 in range(128):

            idx2_8: int32 = idx1_8 + 128
            tw_idx_8: int32 = j8 * 4
            tr_8: float32 = twiddle_real[tw_idx_8]
            ti_8: float32 = twiddle_imag[tw_idx_8]

            ui_8: float32 = y_imag[idx1_8]
            vr_8: float32 = y_real[idx2_8]
            vi_8: float32 = y_imag[idx2_8]
            trv_8: float32 = tr_8 * vr_8 - ti_8 * vi_8
            tiv_8: float32 = tr_8 * vi_8 + ti_8 * vr_8
            y_real[idx1_8] = ur_8 + trv_8
            y_imag[idx1_8] = ui_8 + tiv_8
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# Particionado conservador: solo los arrays cuyo patron de acceso cambia de
# stride entre etapas (y_real/y_imag, con indices base+j y base+j+half donde
# half varia por etapa) necesitan Partition.Complete -- un factor fijo
# Cyclic/Block no puede evitar conflictos de banco para 10 patrones de
# stride distintos. No se particionan twiddle_real/twiddle_imag ni x_real/
# x_imag para mantener el espacio de diseño de la sintesis lo mas pequeno
# posible y evitar el timeout observado en el intento anterior.
s.partition("kernel:y_real", partition_type=Partition.Complete)
s.partition("kernel:y_imag", partition_type=Partition.Complete)

# Pipeline con el objetivo_ii de la especificacion (10) en cada bucle
# independiente, sin unroll adicional, para mantener las directivas al
# minimo y dejar converger al solver de sintesis.

# Tabla de twiddle factors: cada iteracion de k es independiente.
s.pipeline("k", initiation_interval=10)

# Bit-reversal: cada iteracion de i es independiente (el bucle interno b
# es una recurrencia corta de 10 pasos fijos, no se pipelinea).
s.pipeline("i", initiation_interval=10)

# Por cada etapa de la mariposa, se pipelinea SOLO el bucle (g o j) que
# tiene mas iteraciones -- el otro es demasiado pequeno para beneficiarse
# y dejar sin pipeline al bucle dominante era la causa de bucles largos
# sin solapar en el intento anterior.
s.pipeline("g1", initiation_interval=10)   # g1:512  vs j1:1
s.pipeline("g2", initiation_interval=10)   # g2:256  vs j2:2
s.pipeline("g3", initiation_interval=10)   # g3:128  vs j3:4
s.pipeline("g4", initiation_interval=10)   # g4:64   vs j4:8
s.pipeline("g5", initiation_interval=10)   # g5:32   vs j5:16
s.pipeline("j6", initiation_interval=10)   # g6:16   vs j6:32
s.pipeline("j7", initiation_interval=10)   # g7:8    vs j7:64
s.pipeline("j8", initiation_interval=10)   # g8:4    vs j8:128
s.pipeline("j9", initiation_interval=10)   # g9:2    vs j9:256
s.pipeline("j10", initiation_interval=10)  # g10:1   vs j10:512
```
