```python
import allo
from allo.ir.types import float32, int32


def kernel(x_real: float32[1024], x_imag: float32[1024],
           y_real: float32[1024], y_imag: float32[1024]):

    perm: int32[1024]
    buf_real: float32[1024]
    buf_imag: float32[1024]

    # ---- Calcula la tabla de permutacion por digit-reversal base 4 (5 digitos) ----
    for pi_i in range(1024):
        tmp: int32 = pi_i
        rev: int32 = 0
        for pd in range(5):
            digit: int32 = tmp - (tmp / 4) * 4
            rev = rev * 4 + digit
            tmp = tmp / 4
        perm[pi_i] = rev

    # ---- Carga la entrada ya reordenada ----
    for li in range(1024):
        buf_real[li] = x_real[perm[li]]
        buf_imag[li] = x_imag[perm[li]]

    # =========================================================
    # Etapa 0: M=4, M4=1, NUM_GROUPS=256, BASEK=256  (g:256, j:1)
    # =========================================================
    for g0 in range(256):
        for j0 in range(1):
            idx0_s0: int32 = g0 * 4 + j0
            idx1_s0: int32 = idx0_s0 + 1
            idx2_s0: int32 = idx0_s0 + 2
            idx3_s0: int32 = idx0_s0 + 3

            e1_s0: int32 = j0 * 256
            e1m_s0: int32 = e1_s0 - (e1_s0 / 1024) * 1024
            if e1m_s0 > 512:
                e1m_s0 = e1m_s0 - 1024
            e1f_s0: float32 = e1m_s0

            x3_s0: float32 = x2_s0 * ang1_s0
            x4_s0: float32 = x2_s0 * x2_s0
            x5_s0: float32 = x4_s0 * ang1_s0
            x6_s0: float32 = x4_s0 * x2_s0
            x7_s0: float32 = x6_s0 * ang1_s0
            x8_s0: float32 = x4_s0 * x4_s0
            x9_s0: float32 = x8_s0 * ang1_s0

            tw3_re_s0: float32 = tw2_re_s0 * tw1_re_s0 - tw2_im_s0 * tw1_im_s0
            tw3_im_s0: float32 = tw2_re_s0 * tw1_im_s0 + tw2_im_s0 * tw1_re_s0

            a0_re_s0: float32 = buf_real[idx0_s0]
            a0_im_s0: float32 = buf_imag[idx0_s0]
            a1_re_s0: float32 = buf_real[idx1_s0] * tw1_re_s0 - buf_imag[idx1_s0] * tw1_im_s0
            a1_im_s0: float32 = buf_real[idx1_s0] * tw1_im_s0 + buf_imag[idx1_s0] * tw1_re_s0
            a2_re_s0: float32 = buf_real[idx2_s0] * tw2_re_s0 - buf_imag[idx2_s0] * tw2_im_s0
            a2_im_s0: float32 = buf_real[idx2_s0] * tw2_im_s0 + buf_imag[idx2_s0] * tw2_re_s0
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# buf_real/buf_imag se acceden con stride variable (M4 distinto por etapa) --
# se particionan de forma ciclica; para objetivo_ii=5 el factor probado que
# cierra el II real es 8 (ver docs/bitacora.md, fft_radix2_ii5.json).
s.partition("kernel:buf_real", partition_type=Partition.Cyclic, factor=8)
s.partition("kernel:buf_imag", partition_type=Partition.Cyclic, factor=8)

# Por etapa, se pipelinea el bucle (g o j) con MAS iteraciones:
# Etapa0 g:256 j:1   -> g0
# Etapa1 g:64  j:4   -> g1
# Etapa2 g:16  j:16  -> empate, se usa g2
# Etapa3 g:4   j:64  -> j3
# Etapa4 g:1   j:256 -> j4
s.pipeline("g0", initiation_interval=5)
s.pipeline("g1", initiation_interval=5)
s.pipeline("g2", initiation_interval=5)
s.pipeline("j3", initiation_interval=5)
s.pipeline("j4", initiation_interval=5)
```
