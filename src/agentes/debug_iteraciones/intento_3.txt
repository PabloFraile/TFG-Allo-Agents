### KERNEL
import allo
from allo.ir.types import float32, int32


def kernel(x_real: float32[1024], x_imag: float32[1024],
           y_real: float32[1024], y_imag: float32[1024]):
    buf_real: float32[1024]
    buf_imag: float32[1024]

    # ---- base-4 digit-reversal permutation (5 digits, since 4^5 = 1024) ----
    for i in range(1024):
        i0: int32 = i
        q0: int32 = i0 / 4
        d0: int32 = i0 - q0 * 4
        i1: int32 = q0
        q1: int32 = i1 / 4
        d1: int32 = i1 - q1 * 4
        i2: int32 = q1
        q2: int32 = i2 / 4
        d2: int32 = i2 - q2 * 4
        i3: int32 = q2
        q3: int32 = i3 / 4
        d3: int32 = i3 - q3 * 4
        i4: int32 = q3
        q4: int32 = i4 / 4
        d4: int32 = i4 - q4 * 4
        rev: int32 = d0 * 256 + d1 * 64 + d2 * 16 + d3 * 4 + d4
        buf_real[rev] = x_real[i]
        buf_imag[rev] = x_imag[i]

    # ================= STAGE 1: m=4, quarter=1, num_groups=256 =================
    for g1 in range(256):
        for j1 in range(1):
            k1: int32 = g1 * 4
            idx0: int32 = k1 + j1
            idx1: int32 = idx0 + 1
            idx2: int32 = idx1 + 1
            idx3: int32 = idx2 + 1

            s1_ar: float32 = buf_real[idx0]
            s1_ai: float32 = buf_imag[idx0]
            s1_br: float32 = buf_real[idx1]
            s1_bi: float32 = buf_imag[idx1]
            s1_cr: float32 = buf_real[idx2]
            s1_ci: float32 = buf_imag[idx2]
            s1_dr: float32 = buf_real[idx3]
            s1_di: float32 = buf_imag[idx3]

            s1_jf: float32 = j1
            s1_x: float32 = 1.5707963267948966 * s1_jf

            s1_x2: float32 = s1_x * s1_x
            s1_x3: float32 = s1_x2 * s1_x
            s1_x4: float32 = s1_x2 * s1_x2
            s1_x5: float32 = s1_x4 * s1_x
            s1_x6: float32 = s1_x4 * s1_x2
            s1_x7: float32 = s1_x6 * s1_x
            s1_x8: float32 = s1_x4 * s1_x4
            s1_x9: float32 = s1_x8 * s1_x
            s1_x10: float32 = s1_x8 * s1_x2
            s1_x11: float32 = s1_x10 * s1_x

            s1_cos: float32 = 1.0 - s1_x2 / 2.0 + s1_x4 / 24.0 - s1_x6 / 720.0 + s1_x8 / 40320.0 - s1_x10 / 3628800.0
            s1_sin: float32 = s1_x - s1_x3 / 6.0 + s1_x5 / 120.0 - s1_x7 / 5040.0 + s1_x9 / 362880.0 - s1_x11 / 39916800.0

            s1_cos2: float32 = s1_cos * s1_cos - s1_sin * s1_sin
            s1_sin2: float32 = 2.0 * s1_sin * s1_cos
            s1_cos3: float32 = s1_cos2 * s1_cos - s1_sin2 * s1_sin
            s1_sin3: float32 = s1_sin2 * s1_cos + s1_cos2 * s1_sin

            s1_w1r: float32 = s1_cos
            s1_w1i: float32 = -s1_sin
            s1_w2r: float32 = s1_cos2
            s1_w2i: float32 = -s1_sin2
            s1_w3r: float32 = s1_cos3
            s1_w3i: float32 = -s1_sin3

            s1_bpr: float32 = s1_br * s1_w1r - s1_bi * s1_w1i
            s1_bpi: float32 = s1_br * s1_w1i + s1_bi * s1_w1r
            s1_cpr: float32 = s1_cr * s1_w2r - s1_ci * s1_w2i
            s1_cpi: float32 = s1_cr * s1_w2i + s1_ci * s1_w2r
            s1_dpr: float32 = s1_dr * s1_w3r - s1_di * s1_w3i
            s1_dpi: float32 = s1_dr * s1_w3i + s1_di * s1_w3r

            s1_t0r: float32 = s1_ar + s1_cpr
            s1_t0i: float32 = s1_ai + s1_cpi
            s1_t1r: float32 = s1_ar - s1_cpr
            s1_t1i: float32 = s1_ai - s1_cpi
            s1_t2r: float32 = s1_bpr + s1_dpr
            s1_t2i: float32 = s1_bpi + s1_dpi
            s1_t3r: float32 = s1_bpr - s1_dpr
            s1_t3i: float32 = s1_bpi - s1_dpi

            buf_real[idx0] = s1_t0r + s1_t2r
            buf_imag[idx0] = s1_t0i + s1_t2i
            buf_real[idx1] = s1_t1r + s1_t3i
            buf_imag[idx1] = s1_t1i - s1_t3r
            buf_real[idx2] = s1_t0r - s1_t2r
            buf_imag[idx2] = s1_t0i - s1_t2i
            buf_real[idx3] = s1_t1r - s1_t3i
            buf_imag[idx3] = s1_t1i + s1_t3r

    # ================= STAGE 2: m=16, quarter=4, num_groups=64 =================
    for g2 in range(64):
        for j2 in range(4):
            k2: int32 = g2 * 16
            idx0: int32 = k2 + j2
            idx1: int32 = idx0 + 4
            idx2: int32 = idx1 + 4
            idx3: int32 = idx2 + 4

            s2_ar: float32 = buf_real[idx0]
            s2_ai: float32 = buf_imag[idx0]
            s2_br: float32 = buf_real[idx1]
            s2_bi: float32 = buf_imag[idx1]
            s2_cr: float32 = buf_real[idx2]
            s2_ci: float32 = buf_imag[idx2]
            s2_dr: float32 = buf_real[idx3]
            s2_di: float32 = buf_imag[idx3]

            s2_jf: float32 = j2
            s2_x: float32 = 0.39269908169872414 * s2_jf

            s2_x2: float32 = s2_x * s2_x
            s2_x3: float32 = s2_x2 * s2_x
            s2_x4: float32 = s2_x2 * s2_x2
            s2_x5: float32 = s2_x4 * s2_x
            s2_x6: float32 = s2_x4 * s2_x2
            s2_x7: float32 = s2_x6 * s2_x
            s2_x8: float32 = s2_x4 * s2_x4
            s2_x9: float32 = s2_x8 * s2_x
            s2_x10: float32 = s2_x8 * s2_x2
            s2_x11: float32 = s2_x10 * s2_x

            s2_cos: float32 = 1.0 - s2_x2 / 2.0 + s2_x4 / 24.0 - s2_x6 / 720.0 + s2_x8 / 40320.0 - s2_x10 / 3628800.0
            s2_sin: float32 = s2_x - s2_x3 / 6.0 + s2_x5 / 120.0 - s2_x7 / 5040.0 + s2_x9 / 362880.0 - s2_x11 / 39916800.0

            s2_cos2: float32 = s2_cos * s2_cos - s2_sin * s2_sin
            s2_sin2: float32 = 2.0 * s2_sin * s2_cos
            s2_cos3: float32 = s2_cos2 * s2_cos - s2_sin2 * s2_sin
            s2_sin3: float32 = s2_sin2 * s2_cos + s2_cos2 * s2_sin

            s2_w1r: float32 = s2_cos
            s2_w1i: float32 = -s2_sin
            s2_w2r: float32 = s2_cos2
            s2_w2i: float32 = -s2_sin2
            s2_w3r: float32 = s2_cos3
            s2_w3i: float32 = -s2_sin3

            s2_bpr: float32 = s2_br * s2_w1r - s2_bi * s2_w1i
            s2_bpi: float32 = s2_br * s2_w1i + s2_bi * s2_w1r
            s2_cpr: float32 = s2_cr * s2_w2r - s2_ci * s2_w2i
            s2_cpi: float32 = s2_cr * s2_w2i + s2_ci * s2_w2r
            s2_dpr: float32 = s2_dr * s2_w3r - s2_di * s2_w3i
            s2_dpi: float32 = s2_dr * s2_w3i + s2_di * s2_w3r

            s2_t0r: float32 = s2_ar + s2_cpr
            s2_t0i: float32 = s2_ai + s2_cpi
            s2_t1r: float32 = s2_ar - s2_cpr
            s2_t1i: float32 = s2_ai - s2_cpi
            s2_t2r: float32 = s2_bpr + s2_dpr
            s2_t2i: float32 = s2_bpi + s2_dpi
            s2_t3r: float32 = s2_bpr - s2_dpr
            s2_t3i: float32 = s2_bpi - s2_dpi

            buf_real[idx0] = s2_t0r + s2_t2r
            buf_imag[idx0] = s2_t0i + s2_t2i
            buf_real[idx1] = s2_t1r + s2_t3i
            buf_imag[idx1] = s2_t1i - s2_t3r
            buf_real[idx2] = s2_t0r - s2_t2r
            buf_imag[idx2] = s2_t0i - s2_t2i
            buf_real[idx3] = s2_t1r - s2_t3i
            buf_imag[idx3] = s2_t1i + s2_t3r

    # ================= STAGE 3: m=64, quarter=16, num_groups=16 =================
    for g3 in range(16):
        for j3 in range(16):
            k3: int32 = g3 * 64
            idx0: int32 = k3 + j3
            idx1: int32 = idx0 + 16
            idx2: int32 = idx1 + 16
            idx3: int32 = idx2 + 16

            s3_ar: float32 = buf_real[idx0]
            s3_ai: float32 = buf_imag[idx0]
            s3_br: float32 = buf_real[idx1]
            s3_bi: float32 = buf_imag[idx1]
            s3_cr: float32 = buf_real[idx2]
            s3_ci: float32 = buf_imag[idx2]
            s3_dr: float32 = buf_real[idx3]
            s3_di: float32 = buf_imag[idx3]

            s3_jf: float32 = j3
            s3_x: float32 = 0.0981747704246810 * s3_jf

            s3_x2: float32 = s3_x * s3_x
            s3_x3: float32 = s3_x2 * s3_x
            s3_x4: float32 = s3_x2 * s3_x2
            s3_x5: float32 = s3_x4 * s3_x
            s3_x6: float32 = s3_x4 * s3_x2
            s3_x7: float32 = s3_x6 * s3_x
            s3_x8: float32 = s3_x4 * s3_x4
            s3_x9: float32 = s3_x8 * s3_x
            s3_x10: float32 = s3_x8 * s3_x2
            s3_x11: float32 = s3_x10 * s3_x

            s3_cos: float32 = 1.0 - s3_x2 / 2.0 + s3_x4 / 24.0 - s3_x6 / 720.0 + s3_x8 / 40320.0 - s3_x10 / 3628800.0
            s3_sin: float32 = s3_x - s3_x3 / 6.0 + s3_x5 / 120.0 - s3_x7 / 5040.0 + s3_x9 / 362880.0 - s3_x11 / 39916800.0

            s3_cos2: float32 = s3_cos * s3_cos - s3_sin * s3_sin
            s3_sin2: float32 = 2.0 * s3_sin * s3_cos
            s3_cos3: float32 = s3_cos2 * s3_cos - s3_sin2 * s3_sin
            s3_sin3: float32 = s3_sin2 * s3_cos + s3_cos2 * s3_sin

            s3_w1r: float32 = s3_cos
            s3_w1i: float32 = -s3_sin
            s3_w2r: float32 = s3_cos2
            s3_w2i: float32 = -s3_sin2
            s3_w3r: float32 = s3_cos3
            s3_w3i: float32 = -s3_sin3

            s3_bpr: float32 = s3_br * s3_w1r - s3_bi * s3_w1i
            s3_bpi: float32 = s3_br * s3_w1i + s3_bi * s3_w1r
            s3_cpr: float32 = s3_cr * s3_w2r - s3_ci * s3_w2i
            s3_cpi: float32 = s3_cr * s3_w2i + s3_ci * s3_w2r
            s3_dpr: float32 = s3_dr * s3_w3r - s3_di * s3_w3i
            s3_dpi: float32 = s3_dr * s3_w3i + s3_di * s3_w3r

            s3_t0r: float32 = s3_ar + s3_cpr
            s3_t0i: float32 = s3_ai + s3_cpi
            s3_t1r: float32 = s3_ar - s3_cpr
            s3_t1i: float32 = s3_ai - s3_cpi
            s3_t2r: float32 = s3_bpr + s3_dpr
            s3_t2i: float32 = s3_bpi + s3_dpi
            s3_t3r: float32 = s3_bpr - s3_dpr
            s3_t3i: float32 = s3_bpi - s3_dpi

            buf_real[idx0] = s3_t0r + s3_t2r
            buf_imag[idx0] = s3_t0i + s3_t2i
            buf_real[idx1] = s3_t1r + s3_t3i
            buf_imag[idx1] = s3_t1i - s3_t3r
            buf_real[idx2] = s3_t0r - s3_t2r
            buf_imag[idx2] = s3_t0i - s3_t2i
            buf_real[idx3] = s3_t1r - s3_t3i
            buf_imag[idx3] = s3_t1i + s3_t3r

    # ================= STAGE 4: m=256, quarter=64, num_groups=4 =================
    for g4 in range(4):
        for j4 in range(64):
            k4: int32 = g4 * 256
            idx0: int32 = k4 + j4
            idx1: int32 = idx0 + 64
            idx2: int32 = idx1 + 64
            idx3: int32 = idx2 + 64

            s4_ar: float32 = buf_real[idx0]
            s4_ai: float32 = buf_imag[idx0]
            s4_br: float32 = buf_real[idx1]
            s4_bi: float32 = buf_imag[idx1]
            s4_cr: float32 = buf_real[idx2]
            s4_ci: float32 = buf_imag[idx2]
            s4_dr: float32 = buf_real[idx3]
            s4_di: float32 = buf_imag[idx3]

            s4_jf: float32 = j4
            s4_x: float32 = 0.024543692606170259 * s4_jf

            s4_x2: float32 = s4_x * s4_x
            s4_x3: float32 = s4_x2 * s4_x
            s4_x4: float32 = s4_x2 * s4_x2
            s4_x5: float32 = s4_x4 * s4_x
            s4_x6: float32 = s4_x4 * s4_x2
            s4_x7: float32 = s4_x6 * s4_x
            s4_x8: float32 = s4_x4 * s4_x4
            s4_x9: float32 = s4_x8 * s4_x
            s4_x10: float32 = s4_x8 * s4_x2
            s4_x11: float32 = s4_x10 * s4_x

            s4_cos: float32 = 1.0 - s4_x2 / 2.0 + s4_x4 / 24.0 - s4_x6 / 720.0 + s4_x8 / 40320.0 - s4_x10 / 3628800.0
            s4_sin: float32 = s4_x - s4_x3 / 6.0 + s4_x5 / 120.0 - s4_x7 / 5040.0 + s4_x9 / 362880.0 - s4_x11 / 39916800.0

            s4_cos2: float32 = s4_cos * s4_cos - s4_sin * s4_sin

            s4_w1r: float32 = s4_cos
            s4_w1i: float32 = -s4_sin
            s4_w2r: float32 = s4_cos2
            s4_w2i: float32 = -s4_sin2
            s4_w3r: float32 = s4_cos3

            s4_bpr: float32 = s4_br * s4_w1r - s4_bi * s4_w1i
            s4_bpi: float32 = s4_br * s4_w1i + s4_bi * s4_w1r
            s4_cpr: float32 = s4_cr * s4_w2r - s4_ci * s4_w2i
            s4_cpi: float32 = s4_cr * s4_w2i + s4_ci * s4_w2r

            buf_real[idx1] = s4_t1r + s4_t3i
            buf_imag[idx1] = s4_t1i - s4_t3r
            buf_real[idx2] = s4_t0r - s4_t2r
            buf_imag[idx2] = s4_t0i - s4_t2i
            buf_real[idx3] = s4_t1r - s4_t3i
            buf_imag[idx3] = s4_t1i + s4_t3r

    # ================= STAGE 5: m=1024, quarter=256, num_groups=1 =================
    for g5 in range(1):
        for j5 in range(256):
            k5: int32 = g5 * 1024
            idx0: int32 = k5 + j5
            idx1: int32 = idx0 + 256

### SCHEDULE
from allo.customize import Partition

s = allo.customize(kernel)

# Partition the working buffers so the butterfly accesses (which use a
# different stride each stage) can be served concurrently and the
# per-stage loops can pipeline down to the requested II.
s.partition("kernel:buf_real", partition_type=Partition.Cyclic, factor=4)
s.partition("kernel:buf_imag", partition_type=Partition.Cyclic, factor=4)

# Digit-reversal permutation loop.
s.pipeline("i", initiation_interval=5)

# Stage 1 (g1:256, j1:1) -> dominant loop is g1.
s.pipeline("g1", initiation_interval=5)

# Stage 2 (g2:64, j2:4) -> dominant loop is g2.
s.pipeline("g2", initiation_interval=5)

# Stage 3 (g3:16, j3:16) -> tie, pipeline the inner loop j3.
s.pipeline("j3", initiation_interval=5)

# Stage 4 (g4:4, j4:64) -> dominant loop is j4.
s.pipeline("j4", initiation_interval=5)
