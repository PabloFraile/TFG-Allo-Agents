### KERNEL
```python
import allo
from allo.ir.types import float32, int32


def kernel(x_real: float32[1024], x_imag: float32[1024],
           y_real: float32[1024], y_imag: float32[1024]):
    buf_real: float32[1024]
    buf_imag: float32[1024]

    # Bit-reversal permutation (10 bits, since N = 1024 = 2^10)
    for i in range(1024):
        rev: int32 = 0
        idx: int32 = i
        for b in range(10):
            bit: int32 = idx % 2
            idx = idx / 2
            rev = rev * 2 + bit
        buf_real[rev] = x_real[i]
        buf_imag[rev] = x_imag[i]

    # Iterative Cooley-Tukey radix-2 DIT FFT, 10 stages
    m: int32 = 1
    for stage in range(10):
        m = m * 2
        half: int32 = m / 2
        num_groups: int32 = 1024 / m
        for g in range(num_groups):
            for j in range(half):
                j_f: float32 = j
                m_f: float32 = m
                pi_const: float32 = 3.141592653589793
                angle: float32 = -2.0 * pi_const * j_f / m_f

                x2: float32 = angle * angle
                x3: float32 = x2 * angle
                x4: float32 = x2 * x2
                x5: float32 = x4 * angle
                x6: float32 = x4 * x2
                x7: float32 = x6 * angle
                x8: float32 = x4 * x4
                x9: float32 = x8 * angle
                x10: float32 = x8 * x2
                x11: float32 = x10 * angle
                x12: float32 = x6 * x6
                x13: float32 = x12 * angle

                wr: float32 = 1.0 - x2 / 2.0 + x4 / 24.0 - x6 / 720.0 + x8 / 40320.0 - x10 / 3628800.0 + x12 / 479001600.0
                wi: float32 = angle - x3 / 6.0 + x5 / 120.0 - x7 / 5040.0 + x9 / 362880.0 - x11 / 39916800.0 + x13 / 6227020800.0

                base: int32 = g * m
                idx1: int32 = base + j
                idx2: int32 = base + j + half

                or2: float32 = buf_real[idx2]
                oi2: float32 = buf_imag[idx2]

                tr: float32 = or2 * wr - oi2 * wi
                ti: float32 = or2 * wi + oi2 * wr

                ur: float32 = buf_real[idx1]
                ui: float32 = buf_imag[idx1]

                buf_real[idx1] = ur + tr
                buf_imag[idx1] = ui + ti
                buf_real[idx2] = ur - tr
                buf_imag[idx2] = ui - ti

    for k in range(1024):
        y_real[k] = buf_real[k]
        y_imag[k] = buf_imag[k]
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# The outer 'stage' loop has only 10 iterations, but 'half'/'num_groups'
# inside its body depend on 'stage', so it must be fully unrolled to let
# HLS resolve those bounds as compile-time constants.
s.unroll("stage", factor=0)

# 'buf_real'/'buf_imag' are accessed with a stride ('half') that changes
# from copy to copy of the unrolled 'stage' loop (1, 2, 4, ..., 512), so a
# fixed Cyclic/Block factor cannot avoid bank conflicts. Use Complete.
s.partition("kernel:buf_real", partition_type=Partition.Complete)
s.partition("kernel:buf_imag", partition_type=Partition.Complete)

# Pipeline the innermost butterfly loop to hit the target II.
s.pipeline("j", initiation_interval=2)

# Pipeline the simple sequential copy loops for better throughput.
s.pipeline("i", initiation_interval=1)
s.pipeline("k", initiation_interval=1)
```
