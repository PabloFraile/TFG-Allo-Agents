```python
    for i in range(1024):
        idx: int32 = i
        rev: int32 = 0
        for b in range(10):
            half_idx: int32 = idx / 2
            bit: int32 = idx - half_idx * 2
            idx = half_idx
            rev = rev * 2 + bit
        y_real[i] = x_real[rev]
        y_imag[i] = x_imag[rev]

            angle1: float32 = -3.14159265358979 * j1_f
            x2_1: float32 = angle1 * angle1
            x4_1: float32 = x2_1 * x2_1
            x6_1: float32 = x4_1 * x2_1
            x8_1: float32 = x4_1 * x4_1
            x10_1: float32 = x8_1 * x2_1
            cos1: float32 = 1.0 - x2_1 * 0.5 + x4_1 * 0.0416666667 - x6_1 * 0.0013888889 + x8_1 * 0.0000248016 - x10_1 * 0.0000002756
            sin1: float32 = angle1 * (1.0 - x2_1 * 0.1666666667 + x4_1 * 0.0083333333 - x6_1 * 0.0001984127 + x8_1 * 0.0000027557 - x10_1 * 0.0000000251)
            idx1_1: int32 = base1 + j1
            idx2_1: int32 = base1 + j1 + 1
            uReal1: float32 = y_real[idx1_1]
            uImag1: float32 = y_imag[idx1_1]
            vReal1: float32 = y_real[idx2_1]
            vImag1: float32 = y_imag[idx2_1]
            tReal1: float32 = cos1 * vReal1 - sin1 * vImag1
            tImag1: float32 = cos1 * vImag1 + sin1 * vReal1
            y_real[idx1_1] = uReal1 + tReal1
            y_imag[idx1_1] = uImag1 + tImag1
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# Stride entre copias desenrolladas varía de etapa a etapa (half = 1,2,4,...,512)
# => particionado completo obligatorio en los arrays de trabajo.
s.partition("kernel:y_real", partition_type=Partition.Complete)
s.partition("kernel:y_imag", partition_type=Partition.Complete)

# Permutacion bit-reversal
s.pipeline("i", initiation_interval=20)

# Por cada etapa, pipelinear el bucle (g o j) con MAS iteraciones:
s.pipeline("g1", initiation_interval=20)
s.pipeline("g2", initiation_interval=20)
s.pipeline("g3", initiation_interval=20)
s.pipeline("g4", initiation_interval=20)
s.pipeline("g5", initiation_interval=20)
s.pipeline("j6", initiation_interval=20)
s.pipeline("j7", initiation_interval=20)
s.pipeline("j8", initiation_interval=20)
s.pipeline("j9", initiation_interval=20)
s.pipeline("j10", initiation_interval=20)
```
