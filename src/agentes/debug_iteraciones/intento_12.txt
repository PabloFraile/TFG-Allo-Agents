```python
    # Permutacion bit-reversal: y[i] = x[bitreverse(i)]
    for i in range(1024):
        idx: int32 = i
        rev: int32 = 0
        for b in range(10):
            bit: int32 = idx % 2
            idx = idx / 2
            rev = rev * 2 + bit

        y_imag[i] = x_imag[rev]

        for j1 in range(1):
            j1_f: float32 = j1
            angle1: float32 = -2.0 * PI * j1_f / 2.0
            x2_1: float32 = angle1 * angle1
            cos1: float32 = 1.0 - x2_1 / 2.0 + x2_1 * x2_1 / 24.0 - x2_1 * x2_1 * x2_1 / 720.0 + x2_1 * x2_1 * x2_1 * x2_1 / 40320.0
            sin1: float32 = angle1 - angle1 * x2_1 / 6.0 + angle1 * x2_1 * x2_1 / 120.0 - angle1 * x2_1 * x2_1 * x2_1 / 5040.0 + angle1 * x2_1 * x2_1 * x2_1 * x2_1 / 362880.0
            idx_a1: int32 = base1 + j1
            idx_b1: int32 = base1 + j1 + 1

            ui1: float32 = y_imag[idx_a1]
            vr1: float32 = y_real[idx_b1]
            vi1: float32 = y_imag[idx_b1]
            tr1: float32 = cos1 * vr1 - sin1 * vi1
            ti1: float32 = cos1 * vi1 + sin1 * vr1
```

### SCHEDULE
```python
from allo.customize import Partition

s = allo.customize(kernel)

# El array de salida sufre patrones de acceso con stride variable
# (half cambia de etapa en etapa: 1,2,4,...,512) entre las 10 copias
# desenrolladas, asi que se particiona por completo para evitar
# conflictos de banco de memoria.
s.partition("kernel:y_real", partition_type=Partition.Complete)
s.partition("kernel:y_imag", partition_type=Partition.Complete)

# En cada etapa se pipelinea el bucle que tiene MAS iteraciones
# (el que realmente domina el numero de ciclos), no siempre el mismo
# eje, siguiendo la tabla g:num_groups / j:half de cada etapa.
s.pipeline("g1", initiation_interval=20)
s.pipeline("g2", initiation_interval=20)
s.pipeline("g3", initiation_interval=20)
s.pipeline("g4", initiation_interval=20)
s.pipeline("g5", initiation_interval=20)
s.pipeline("j6", initiation_interval=20)
s.pipeline("j7", initiation_interval=20)
s.pipeline("j8", initiation_interval=20)
s.pipeline("j9", initiation_interval=20)
s.pipeline("j10", initiation_interval=20)
```
